package rules

// Oracle for the code-review-ocr protocol port (scratch, never upstreamed).

import (
	"encoding/json"
	"os"
	"path/filepath"
	"testing"

	allowedext "github.com/alibaba/open-code-review/internal/config/allowlist"
)

type repoCase struct {
	Files  map[string]string `json:"files"`
	Custom string            `json:"custom"`
	Paths  []string          `json:"paths"`
}

func TestOracle(t *testing.T) {
	in := os.Getenv("OCR_ORACLE_IN")
	if in == "" {
		t.Skip("no oracle input")
	}
	raw, err := os.ReadFile(in)
	if err != nil {
		t.Fatal(err)
	}
	var cases struct {
		Paths []string   `json:"paths"`
		Repos []repoCase `json:"repos"`
	}
	if err := json.Unmarshal(raw, &cases); err != nil {
		t.Fatal(err)
	}
	t.Setenv("HOME", t.TempDir())

	sys, err := LoadDefault()
	if err != nil {
		t.Fatal(err)
	}
	type pathOut struct {
		Secret   bool   `json:"secret"`
		Allowed  bool   `json:"allowed"`
		Excluded bool   `json:"excluded"`
		Pattern  string `json:"pattern"`
	}
	var paths []pathOut
	for _, p := range cases.Paths {
		ext := ""
		base := filepath.Base(p)
		if dot := lastDot(base); dot > 0 {
			ext = toLower(base[dot:])
		}
		paths = append(paths, pathOut{
			Secret:   allowedext.IsSecretPath(p),
			Allowed:  ext == "" || allowedext.IsAllowedExt(ext),
			Excluded: allowedext.IsExcludedPath(p),
			Pattern:  sys.resolveDetail(p).Pattern,
		})
	}

	type repoOut struct {
		Rules   []string    `json:"rules"`
		Filter  *FileFilter `json:"filter"`
	}
	var repos []repoOut
	for _, rc := range cases.Repos {
		dir := t.TempDir()
		for rel, content := range rc.Files {
			full := filepath.Join(dir, rel)
			_ = os.MkdirAll(filepath.Dir(full), 0o755)
			_ = os.WriteFile(full, []byte(content), 0o644)
		}
		custom := ""
		if rc.Custom != "" {
			custom = filepath.Join(dir, rc.Custom)
		}
		res, filter, err := NewResolver(dir, custom, ResolverOptions{})
		if err != nil {
			repos = append(repos, repoOut{Rules: []string{"ERROR: " + err.Error()}})
			continue
		}
		var texts []string
		for _, p := range rc.Paths {
			texts = append(texts, res.Resolve(p))
		}
		repos = append(repos, repoOut{Rules: texts, Filter: filter})
	}
	data, _ := json.MarshalIndent(map[string]any{"paths": paths, "repos": repos}, "", " ")
	if err := os.WriteFile(os.Getenv("OCR_ORACLE_OUT"), data, 0o644); err != nil {
		t.Fatal(err)
	}
}

func lastDot(s string) int {
	for i := len(s) - 1; i >= 0; i-- {
		if s[i] == '.' {
			return i
		}
	}
	return -1
}

func toLower(s string) string {
	b := []byte(s)
	for i, c := range b {
		if c >= 'A' && c <= 'Z' {
			b[i] = c + 32
		}
	}
	return string(b)
}
